import { expect, it } from 'vitest';
import { Pool } from 'pg';
import { newId, now, platformConnectionSchema } from '@ace/shared';
import { MemoryRepository, PgRepository } from '@ace/database';
import { ensureChannel } from '@ace/channel-brain';
import { TokenVault, ConnectionService } from '@ace/publisher/auth';

const url = process.env.DATABASE_URL;
const conn = (channelId: string, token: string, expiresAt: string | null, vault: TokenVault) => platformConnectionSchema.parse({
  id: newId(), channelId, destination: 'youtube_long', externalAccountId: 'UC-same-account', displayName: 'Same', scopes: [],
  connectedAt: now(), expiresAt, encryptedCredentials: vault.seal({ accessToken: token, refreshToken: null, expiresAt, scopes: [] }) });

it.skipIf(!url)('PG1 reconnecting the same platform account fails with a unique violation in Postgres', async () => {
  const pool = new Pool({ connectionString: url }); const repository = new PgRepository(pool); const channelId = newId();
  const vault = new TokenVault(Buffer.alloc(32, 1));
  try {
    await ensureChannel(repository, channelId);
    await repository.put('platform_connections', conn(channelId, 'old', new Date(Date.now() - 1000).toISOString(), vault));
    await expect(repository.put('platform_connections', conn(channelId, 'new', null, vault))).rejects.toThrow(/duplicate key|unique/i);
  } finally {
    for (const t of ['platform_connections', 'audit_events']) await pool.query(`DELETE FROM ${t} WHERE channel_id = $1`, [channelId]);
    await pool.query('DELETE FROM channels WHERE id = $1', [channelId]); await pool.end();
  }
});

it('PG1b memory: after reconnect the first (expired) connection is still selected', async () => {
  const repository = new MemoryRepository(); const channelId = newId(); await ensureChannel(repository, channelId);
  const vault = new TokenVault(Buffer.alloc(32, 1));
  const old = conn(channelId, 'old', new Date(Date.now() - 1000).toISOString(), vault);
  await repository.put('platform_connections', old);
  await repository.put('platform_connections', conn(channelId, 'new', null, vault));
  const selected = (await repository.list('platform_connections', channelId)).find(r => r.destination === 'youtube_long')!;
  expect(selected.id).toBe(old.id);
  await expect(new ConnectionService(repository, vault).accessToken(selected)).rejects.toThrow('reconnect account');
});

it.skipIf(!url)('PG2 same-channel lock waiters hold pool connections and stall unrelated reads', async () => {
  const pool = new Pool({ connectionString: url, max: 3 }); const repository = new PgRepository(pool);
  const a = newId(), b = newId();
  try {
    await ensureChannel(repository, a); await ensureChannel(repository, b);
    let release!: () => void; const gate = new Promise<void>(r => { release = r; });
    const holder = repository.withChannelLock(a, () => gate);
    await new Promise(r => setTimeout(r, 100));
    const waiters = [1, 2].map(() => repository.withChannelLock(a, async () => undefined));
    await new Promise(r => setTimeout(r, 200));
    const started = Date.now();
    const unrelated = repository.get('channels', b).then(() => Date.now() - started);
    const stalled = await Promise.race([unrelated, new Promise<string>(r => setTimeout(() => r('stalled'), 1500))]);
    expect(stalled).toBe('stalled');
    release(); await holder; await Promise.all(waiters);
    expect(await unrelated).toBeGreaterThanOrEqual(1400);
  } finally {
    await pool.query('DELETE FROM channels WHERE id = ANY($1::uuid[])', [[a, b]]); await pool.end();
  }
});
